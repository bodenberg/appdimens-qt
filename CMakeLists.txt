cmake_minimum_required(VERSION 3.16)
project(appdimens-qt VERSION 3.2.0 LANGUAGES C CXX)

set(APPDIMENS_QT_MAJOR "" CACHE STRING "Force Qt major version (5 or 6); empty = prefer 6")
option(APPDIMENS_QT_BUILD_TESTS "Build tests" ON)
option(APPDIMENS_QT_BUILD_EXAMPLES "Build examples" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

if(APPDIMENS_QT_MAJOR)
  set(QT_MAJOR ${APPDIMENS_QT_MAJOR})
else()
  find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
  set(QT_MAJOR ${QT_VERSION_MAJOR})
endif()
find_package(Qt${QT_MAJOR} 5.12 REQUIRED COMPONENTS Core Gui)
find_package(Qt${QT_MAJOR} QUIET COMPONENTS Widgets Qml Quick Test)

add_library(appdimens_qt
  include/appdimens/appdimens.h include/appdimens/appdimens_qt.hpp src/appdimens_qt.cpp)
target_include_directories(appdimens_qt PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(appdimens_qt PUBLIC Qt${QT_MAJOR}::Core Qt${QT_MAJOR}::Gui)
if(TARGET Qt${QT_MAJOR}::Widgets)
  target_link_libraries(appdimens_qt PUBLIC Qt${QT_MAJOR}::Widgets)
  target_compile_definitions(appdimens_qt PUBLIC APPDIMENS_QT_WIDGETS)
endif()
if(TARGET Qt${QT_MAJOR}::Qml)
  target_sources(appdimens_qt PRIVATE src/appdimens_qml.cpp)
  target_link_libraries(appdimens_qt PUBLIC Qt${QT_MAJOR}::Qml)
  target_compile_definitions(appdimens_qt PUBLIC APPDIMENS_QT_QML)
endif()
if(MSVC)
  target_compile_options(appdimens_qt PRIVATE /W4 /permissive-)
else()
  target_compile_options(appdimens_qt PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(AppDimens::Qt ALIAS appdimens_qt)

if(APPDIMENS_QT_BUILD_TESTS AND TARGET Qt${QT_MAJOR}::Test)
  enable_testing()
  add_executable(test_core tests/test_core.c)
  target_include_directories(test_core PRIVATE include tests)
  if(NOT MSVC)
    target_link_libraries(test_core m)
  endif()
  add_test(NAME core COMMAND test_core)
  add_executable(test_custom tests/test_custom.c)
  target_include_directories(test_custom PRIVATE include/appdimens tests)
  if(NOT MSVC)
    target_link_libraries(test_custom m)
  endif()
  add_test(NAME custom COMMAND test_custom)
  add_executable(tst_appdimens_qt tests/tst_appdimens_qt.cpp)
  target_link_libraries(tst_appdimens_qt appdimens_qt Qt${QT_MAJOR}::Test)
  if(TARGET Qt${QT_MAJOR}::Quick)
    target_link_libraries(tst_appdimens_qt Qt${QT_MAJOR}::Quick)
    target_compile_definitions(tst_appdimens_qt PRIVATE HAVE_QUICK)
  endif()
  add_test(NAME qt COMMAND tst_appdimens_qt)
  set_tests_properties(qt PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()

if(APPDIMENS_QT_BUILD_EXAMPLES)
  if(TARGET Qt${QT_MAJOR}::Widgets)
    add_executable(appdimens-qt-widgets examples/widgets/main.cpp)
    target_link_libraries(appdimens-qt-widgets appdimens_qt)
  endif()
  if(TARGET Qt${QT_MAJOR}::Quick)
    add_executable(appdimens-qt-qml examples/qml/main.cpp)
    target_link_libraries(appdimens-qt-qml appdimens_qt Qt${QT_MAJOR}::Quick)
    target_compile_definitions(appdimens-qt-qml PRIVATE QML_FILE="${CMAKE_CURRENT_SOURCE_DIR}/examples/qml/main.qml")
  endif()
endif()

include(GNUInstallDirs)
install(TARGETS appdimens_qt EXPORT AppDimensQtTargets)
install(DIRECTORY include/appdimens DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT AppDimensQtTargets NAMESPACE AppDimens:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/AppDimensQt)
